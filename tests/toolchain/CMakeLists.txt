cmake_minimum_required(VERSION 3.25)
project(win16style_toolchain_probe LANGUAGES C CXX)
if(NOT MSVC)
  message(FATAL_ERROR "This probe qualifies the MSVC product profile only")
endif()
if(NOT PROBE_ARCH MATCHES "^(x86|x64)$")
  message(FATAL_ERROR "Select PROBE_ARCH=x86 or x64")
endif()
if((PROBE_ARCH STREQUAL "x86" AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
   OR (PROBE_ARCH STREQUAL "x64" AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8))
  message(FATAL_ERROR "Compiler environment does not match requested architecture")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(probe SHARED dll.c sdk.cpp)
add_executable(probe-run main.c)
foreach(target probe probe-run)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
endforeach()
target_link_libraries(probe-run PRIVATE probe)
enable_testing()
add_test(NAME toolchain/${PROBE_ARCH}/sdk-dll-tls COMMAND probe-run)
set_tests_properties(toolchain/${PROBE_ARCH}/sdk-dll-tls PROPERTIES TIMEOUT 10)
