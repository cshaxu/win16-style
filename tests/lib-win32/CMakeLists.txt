cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(shared_win32_tests C)
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory(process-hook/portable)

add_library(shared_win32_test_providers STATIC
  "${ROOT}/src/lib-win32/handle/duplicate.c"
  "${ROOT}/src/lib-win32/process-image/image.c"
  "${ROOT}/src/lib-win32/process-image/file_handle.c"
  "${ROOT}/src/lib-win32/process-image/remote.c"
  "${ROOT}/src/lib-win32/process-launch/process.c"
  "${ROOT}/src/lib-win32/process-launch/command_line.c")
target_include_directories(shared_win32_test_providers PUBLIC "${ROOT}/src")
target_compile_definitions(shared_win32_test_providers PUBLIC _WIN32_WINNT=0x0A00)
set(shared_tests)
foreach(fixture handle/handle process-image/image process-image/access process-launch/attributes
    process-launch/command_line process-launch/process)
  string(REPLACE "/" "-" name "shared-${fixture}")
  add_executable(${name} "${fixture}.c")
  target_link_libraries(${name} PRIVATE shared_win32_test_providers)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
  list(APPEND shared_tests ${name})
endforeach()
foreach(fixture link payload payload_test payload_direct)
  set(name "shared-hook-${fixture}")
  add_executable(${name} "process-hook/${fixture}.c")
  target_link_libraries(${name} PRIVATE process_hook)
  if(fixture STREQUAL "payload_direct")
    if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
      target_link_options(${name} PRIVATE -municode)
    endif()
    add_test(NAME ${name} COMMAND ${name} "$<TARGET_FILE:${name}>")
  else()
    add_test(NAME ${name} COMMAND ${name})
  endif()
  set_tests_properties(${name} PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  list(APPEND shared_tests ${name})
endforeach()
add_executable(shared-hook-race process-hook/late_guard_race.c
  "${ROOT}/src/lib-win32/process-hook/late_guard.c")
target_include_directories(shared-hook-race PRIVATE "${ROOT}/src")
target_compile_definitions(shared-hook-race PRIVATE PROCESS_HOOK_LATE_GUARD_TEST _WIN32_WINNT=0x0A00)
add_test(NAME shared-hook-race COMMAND shared-hook-race)
set_tests_properties(shared-hook-race PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
add_executable(shared-image-io-errors process-image/io_errors.c)
target_compile_definitions(shared-image-io-errors PRIVATE _WIN32_WINNT=0x0A00)
add_test(NAME shared-image-io-errors COMMAND shared-image-io-errors)
set_tests_properties(shared-image-io-errors PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
list(APPEND shared_tests shared-image-io-errors)
add_executable(shared-remote-memory-errors process-global/remote_memory.c)
add_test(NAME shared-remote-memory-errors COMMAND shared-remote-memory-errors)
set_tests_properties(shared-remote-memory-errors PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
list(APPEND shared_tests shared-remote-memory-errors)
add_executable(shared-payload-memory-errors process-hook/payload_memory.c)
add_test(NAME shared-payload-memory-errors COMMAND shared-payload-memory-errors)
set_tests_properties(shared-payload-memory-errors PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
list(APPEND shared_tests shared-payload-memory-errors)
add_executable(shared-image-machine process-image/image_machine_test.c)
target_link_libraries(shared-image-machine PRIVATE shared_win32_test_providers)
add_executable(shared-cross-payload process-hook/cross_payload.c)
target_link_libraries(shared-cross-payload PRIVATE process_hook)
add_library(shared-cross-helper SHARED process-hook/cross_helper.c
  process-hook/cross_helper.def)
set_target_properties(shared-cross-helper PROPERTIES PREFIX "" OUTPUT_NAME "cross-helper")
target_link_libraries(shared-cross-helper PRIVATE process_hook)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_link_options(shared-cross-payload PRIVATE -municode)
  target_link_options(shared-image-machine PRIVATE -municode)
endif()
add_custom_target(shared-win32-tests DEPENDS ${shared_tests} shared-hook-race
  shared-image-machine shared-cross-payload shared-cross-helper unwind late_batch propagate)
